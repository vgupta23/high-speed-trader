# datetime, json, os, platform, subprocess, sys, time, pathlib, zoneinfo).
# The one gap is Windows, which ships no IANA timezone database for
# zoneinfo to read -- tzdata supplies it there. On macOS/Linux this line
# installs a harmless no-op-equivalent package; pip will skip it entirely
# if your platform doesn't need it.
tzdata; sys_platform == "win32"
